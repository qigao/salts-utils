cmake_minimum_required(VERSION 3.27)
project(SaltsUnicodeCutoverConsumer LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SaltsUtils CONFIG REQUIRED)
foreach(_target IN ITEMS Salts::Core Salts::Unicode Salts::JinjaCMeta)
  if(NOT TARGET ${_target})
    message(FATAL_ERROR "Installed candidate is missing ${_target}")
  endif()
endforeach()
get_target_property(_unicode_imported Salts::Unicode IMPORTED)
if(NOT _unicode_imported)
  message(FATAL_ERROR "Unicode was not imported from the installed Salts SDK")
endif()
add_executable(unicode_cutover_c11 main.c)
target_link_libraries(unicode_cutover_c11 PRIVATE Salts::Unicode Salts::JinjaCMeta)
add_executable(unicode_cutover_cpp17 main.cpp)
target_link_libraries(unicode_cutover_cpp17 PRIVATE Salts::Unicode Salts::JinjaCMeta)
enable_testing()
add_test(NAME unicode_cutover_c11 COMMAND unicode_cutover_c11)
add_test(NAME unicode_cutover_cpp17 COMMAND unicode_cutover_cpp17)
