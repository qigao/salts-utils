cmake_minimum_required(VERSION 3.27)
project(SaltsUnicodePairedInstalledConsumer LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT IS_DIRECTORY "$ENV{SALTS_ROOT}" OR
   NOT IS_DIRECTORY "$ENV{SALTS_UTILS_CANDIDATE_ROOT}")
  message(FATAL_ERROR "Both official Salts and candidate SaltsUtils SDK roots required")
endif()
unset(Salts_DIR CACHE)
find_package(Salts CONFIG REQUIRED
  PATHS "$ENV{SALTS_ROOT}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
unset(SaltsUtils_DIR CACHE)
find_package(SaltsUtils CONFIG REQUIRED
  PATHS "$ENV{SALTS_UTILS_CANDIDATE_ROOT}"
  NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
foreach(_target IN ITEMS Salts::Core Salts::Unicode Salts::JinjaCMeta)
  if(NOT TARGET ${_target})
    message(FATAL_ERROR "Paired installed SDK lacks ${_target}")
  endif()
endforeach()
get_target_property(_from_salts Salts::Unicode IMPORTED)
if(NOT _from_salts)
  message(FATAL_ERROR "Unicode must be imported, never provided by SaltsUtils")
endif()
get_target_property(_unicode_location Salts::Unicode IMPORTED_LOCATION_RELEASE)
if(NOT _unicode_location)
  get_target_property(_unicode_location Salts::Unicode IMPORTED_LOCATION)
endif()
if(NOT _unicode_location)
  message(FATAL_ERROR "Imported Unicode SDK target lacks a real library")
endif()
file(TO_CMAKE_PATH "$ENV{SALTS_ROOT}" _salts_prefix)
cmake_path(IS_PREFIX _salts_prefix "${_unicode_location}"
           NORMALIZE _unicode_from_official)
if(NOT _unicode_from_official)
  message(FATAL_ERROR "Unicode imported library is NOT owned by official Salts SDK")
endif()

add_executable(unicode_paired_c11 main.c)
target_link_libraries(unicode_paired_c11 PRIVATE Salts::Unicode Salts::JinjaCMeta)
add_executable(unicode_paired_cpp17 main.cpp)
target_link_libraries(unicode_paired_cpp17 PRIVATE Salts::Unicode Salts::JinjaCMeta)
enable_testing()
add_test(NAME unicode_paired_c11 COMMAND unicode_paired_c11)
add_test(NAME unicode_paired_cpp17 COMMAND unicode_paired_cpp17)
